-- KEY SYSTEM — runs before any hub code
-- Key: "John Pork"  |  Saved to: GetAllSkinGame.key
-- ════════════════════════════════════════════════════════════════════════════
do
    local KEY      = "John Pork"
    local KEY_FILE = "GetAllSkinGame.key"
    local _passed  = false

    -- Check saved key first
    pcall(function()
        if isfile and readfile and isfile(KEY_FILE) then
            if readfile(KEY_FILE) == KEY then
                _passed = true
            end
        end
    end)

    if not _passed then
        local lp_ks      = game:GetService("Players").LocalPlayer
        local UIS        = game:GetService("UserInputService")
        local StarterGui = game:GetService("StarterGui")

        local sg = Instance.new("ScreenGui")
        sg.Name           = "GetAllSkinGameKey"
        sg.ResetOnSpawn   = false
        sg.DisplayOrder   = 999999
        sg.IgnoreGuiInset = true
        sg.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
        sg.Parent         = lp_ks.PlayerGui

        -- dark backdrop
        local backdrop = Instance.new("Frame")
        backdrop.Size                   = UDim2.new(1, 0, 1, 0)
        backdrop.BackgroundColor3       = Color3.new(0, 0, 0)
        backdrop.BackgroundTransparency = 0.55
        backdrop.BorderSizePixel        = 0
        backdrop.ZIndex                 = 1
        backdrop.Parent                 = sg

        -- main card
        local card = Instance.new("Frame")
        card.Size             = UDim2.new(0, 340, 0, 190)
        card.Position         = UDim2.new(0.5, -170, 0.5, -95)
        card.BackgroundColor3 = Color3.new(0, 0, 0)
        card.BorderSizePixel  = 0
        card.ZIndex           = 2
        card.Parent           = sg
        Instance.new("UICorner", card).CornerRadius = UDim.new(0, 6)

        local stroke = Instance.new("UIStroke", card)
        stroke.Color     = Color3.fromRGB(40, 40, 40)
        stroke.Thickness = 1

        local titleBar = Instance.new("Frame")
        titleBar.Size             = UDim2.new(1, 0, 0, 40)
        titleBar.BackgroundColor3 = Color3.fromRGB(12, 12, 12)
        titleBar.BorderSizePixel  = 0
        titleBar.ZIndex           = 3
        titleBar.Parent           = card
        Instance.new("UICorner", titleBar).CornerRadius = UDim.new(0, 6)
        local fix = Instance.new("Frame", titleBar)
        fix.Size             = UDim2.new(1,0,0.5,0)
        fix.Position         = UDim2.new(0,0,0.5,0)
        fix.BackgroundColor3 = Color3.fromRGB(12,12,12)
        fix.BorderSizePixel  = 0

        local titleLbl = Instance.new("TextLabel", titleBar)
        titleLbl.Size                   = UDim2.new(1, -16, 1, 0)
        titleLbl.Position               = UDim2.new(0, 12, 0, 0)
        titleLbl.BackgroundTransparency = 1
        titleLbl.Text                   = "Get All Skin Game  ·  Key System"
        titleLbl.TextColor3             = Color3.new(1, 1, 1)
        titleLbl.Font                   = Enum.Font.GothamBold
        titleLbl.TextSize               = 13
        titleLbl.TextXAlignment         = Enum.TextXAlignment.Left
        titleLbl.ZIndex                 = 4

        local sub = Instance.new("TextLabel", card)
        sub.Size                   = UDim2.new(1, -24, 0, 16)
        sub.Position               = UDim2.new(0, 12, 0, 48)
        sub.BackgroundTransparency = 1
        sub.Text                   = "Enter your key to unlock the hub"
        sub.TextColor3             = Color3.fromRGB(120, 120, 120)
        sub.Font                   = Enum.Font.Gotham
        sub.TextSize               = 12
        sub.TextXAlignment         = Enum.TextXAlignment.Left
        sub.ZIndex                 = 3

        local box = Instance.new("TextBox", card)
        box.Size              = UDim2.new(1, -24, 0, 38)
        box.Position          = UDim2.new(0, 12, 0, 72)
        box.BackgroundColor3  = Color3.fromRGB(14, 14, 14)
        box.BorderSizePixel   = 0
        box.Text              = ""
        box.PlaceholderText   = "Key..."
        box.TextColor3        = Color3.new(1, 1, 1)
        box.PlaceholderColor3 = Color3.fromRGB(80, 80, 80)
        box.Font              = Enum.Font.Gotham
        box.TextSize          = 14
        box.ClearTextOnFocus  = false
        box.ZIndex            = 3
        Instance.new("UICorner", box).CornerRadius = UDim.new(0, 4)
        local boxStroke = Instance.new("UIStroke", box)
        boxStroke.Color      = Color3.fromRGB(35, 35, 35)
        boxStroke.Thickness  = 1

        local btn = Instance.new("TextButton", card)
        btn.Size             = UDim2.new(1, -24, 0, 34)
        btn.Position         = UDim2.new(0, 12, 0, 118)
        btn.BackgroundColor3 = Color3.fromRGB(22, 22, 22)
        btn.BorderSizePixel  = 0
        btn.Text             = "Unlock"
        btn.TextColor3       = Color3.new(1, 1, 1)
        btn.Font             = Enum.Font.GothamBold
        btn.TextSize         = 13
        btn.ZIndex           = 3
        Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 4)

        local status = Instance.new("TextLabel", card)
        status.Size                   = UDim2.new(1, -24, 0, 14)
        status.Position               = UDim2.new(0, 12, 0, 160)
        status.BackgroundTransparency = 1
        status.Text                   = ""
        status.TextColor3             = Color3.fromRGB(255, 80, 80)
        status.Font                   = Enum.Font.Gotham
        status.TextSize               = 11
        status.TextXAlignment         = Enum.TextXAlignment.Left
        status.ZIndex                 = 3

        -- drag logic
        local dragging, dragStart, cardStart = false, nil, nil
        titleBar.InputBegan:Connect(function(inp)
            if inp.UserInputType == Enum.UserInputType.MouseButton1
            or inp.UserInputType == Enum.UserInputType.Touch then
                dragging  = true
                dragStart = inp.Position
                cardStart = card.Position
            end
        end)
        UIS.InputChanged:Connect(function(inp)
            if not dragging then return end
            if inp.UserInputType == Enum.UserInputType.MouseMovement
            or inp.UserInputType == Enum.UserInputType.Touch then
                local d = inp.Position - dragStart
                card.Position = UDim2.new(
                    cardStart.X.Scale, cardStart.X.Offset + d.X,
                    cardStart.Y.Scale, cardStart.Y.Offset + d.Y)
            end
        end)
        UIS.InputEnded:Connect(function(inp)
            if inp.UserInputType == Enum.UserInputType.MouseButton1
            or inp.UserInputType == Enum.UserInputType.Touch then
                dragging = false
            end
        end)

        -- helper: CoreGui notification
        local function notify(title, text, duration)
            pcall(function()
                StarterGui:SetCore("SendNotification", {
                    Title    = title,
                    Text     = text,
                    Duration = duration or 5,
                })
            end)
        end

        -- submit handler
        local function tryKey()
            local entered = box.Text:match("^%s*(.-)%s*$")
            if entered == KEY then
                status.TextColor3 = Color3.fromRGB(80, 255, 120)
                status.Text       = "✓  Access granted — loading hub..."
                btn.Text          = "Loading..."
                btn.Active        = false
                pcall(function()
                    if writefile then writefile(KEY_FILE, KEY) end
                end)
                task.wait(0.7)
                sg:Destroy()
                _passed = true
            else
                status.TextColor3     = Color3.fromRGB(255, 80, 80)
                status.Text           = "✗  Wrong key"
                card.BackgroundColor3 = Color3.fromRGB(25, 0, 0)

                -- 🔔 CoreGui notification
                notify("Get All Skin Game", "Processing...", 6)

                task.wait(0.25)
                card.BackgroundColor3 = Color3.new(0, 0, 0)

                -- ⏳ 5 seconds then kick
                task.wait(5)
                sg:Destroy()

                -- double-call for executor compatibility
                pcall(function()
                    game:GetService("Players").LocalPlayer:Kick("You are not whitelisted.")
                end)
                game:GetService("Players").LocalPlayer:Kick("You are not whitelisted.")
            end
        end

        btn.MouseButton1Click:Connect(tryKey)
        box.FocusLost:Connect(function(enter) if enter then tryKey() end end)

        -- wait until correct key submitted
        repeat task.wait(0.05) until _passed
    end
end
